cmake_minimum_required(VERSION 3.20)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_LIST_DIR}/../minimal-board/arm-none-eabi.cmake"
        CACHE FILEPATH "ARM bare-metal toolchain")
endif()
project(sim_can_motor LANGUAGES C)
option(SIM_ROTOR_LOOP_TEST "Drive a rigid rotor, then command zero after feedback" OFF)
option(SIM_JOINT_LOOP_TEST "Drive two shared native joints, then command zero after both feedback reads" OFF)
if(SIM_ROTOR_LOOP_TEST AND SIM_JOINT_LOOP_TEST)
    message(FATAL_ERROR "SIM_ROTOR_LOOP_TEST and SIM_JOINT_LOOP_TEST are mutually exclusive")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../minimal-board/firmware.cmake")
if(SIM_ROTOR_LOOP_TEST)
    target_compile_definitions(firmware PRIVATE SIM_ROTOR_LOOP_TEST=1)
endif()
if(SIM_JOINT_LOOP_TEST)
    target_compile_definitions(firmware PRIVATE SIM_JOINT_LOOP_TEST=1)
endif()
